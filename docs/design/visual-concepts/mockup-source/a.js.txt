(() => {
  const st = document.querySelector('.stage')
  const env = document.createElement('div'); env.className = 'env'; env.innerHTML = '<i class="env-side"></i><i class="env-vig"></i>'
  st.insertBefore(env, st.firstChild)
  // the photographer's lighting plan, per editorial chapter: [pool level, pool size, wall wash, vignette, tint]
  const PLAN = [null,
    [0.22, 1.25, 0.15, 0.6, '232,228,221'], // 01 the hero shot: one big soft key, warm-neutral
    [0.22, 1.15, 0.15, 0.6, '232,228,221'], // 02 exterior pass
    [0.2, 1.0, 0.12, 0.64, '226,229,232'],  // 03-05 the bag opens: tighter, a touch cooler (the cold inside)
    [0.2, 1.0, 0.12, 0.64, '226,229,232'],
    [0.2, 1.0, 0.12, 0.64, '226,229,232'],
    [0.17, 1.5, 0.08, 0.6, '230,229,226'],  // 06 looking down: the floor becomes the whole backdrop, a broad even pool
    [0.13, 0.9, 0.08, 0.72, '232,226,218'], // 07 display close-up: low key, the screen and the red carry the frame
    [0.15, 1.0, 0.19, 0.5, '232,228,221'],  // 08 the complete system: the full studio, wide and even
  ]
  window.__envUpdate = (g) => {
    const s = document.documentElement.style, L = PLAN[g.ch]
    const [px, py, rx, ry] = g.ch === 8 ? g.poolWide : g.pool
    s.setProperty('--pool-x', px + 'px'); s.setProperty('--pool-y', py + 'px'); s.setProperty('--pool-rx', Math.round(rx * L[1]) + 'px'); s.setProperty('--pool-ry', Math.round(ry * L[1]) + 'px')
    s.setProperty('--pool-a', L[0]); s.setProperty('--wash-a', L[2]); s.setProperty('--vig-a', L[3]); s.setProperty('--tint', L[4])
    const bx = (g.bbox[0] + g.bbox[2]) / 2, coveY = g.cove[1]
    s.setProperty('--wash-x', Math.round(bx) + 'px'); s.setProperty('--wash-y', Math.round(Math.min(coveY, g.bbox[3]) - innerHeight * 0.16) + 'px')
    s.setProperty('--wash-rx', Math.round(innerWidth * (g.ch === 8 ? 0.5 : 0.36)) + 'px'); s.setProperty('--wash-ry', Math.round(innerHeight * 0.5) + 'px')
    s.setProperty('--cove', coveY + 'px')
  }
})()
