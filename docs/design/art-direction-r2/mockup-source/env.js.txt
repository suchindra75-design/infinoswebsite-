// MOCKUP ENGINE (scratchpad only — injected into the production build for screenshots; the site itself is untouched).
// One studio environment behind the product, driven every frame by the site's own camera maths (window.__CORE):
// a cyclorama (wall, cove, floor) whose cove line is projected from the camera, a floor light pool in true
// perspective, a wall wash behind the product, edge falloff, a mocked contact shadow for the live chapters (the live
// scene's own shadow does not render today), and frame-edge extension so the photoreal shadows never end in a hard
// line. Each concept supplies window.__CONCEPT: palettes and a lighting plan keyed to story progress.
(() => {
  const C = window.__CORE, K = window.__CONCEPT
  const CH = [['intro', 0, 0.01], ['reveal', 0.01, 0.11], ['hero', 0.11, 0.19], ['detail', 0.19, 0.31], ['open', 0.31, 0.57], ['interior', 0.57, 0.69], ['display', 0.69, 0.83], ['overview', 0.83, 1]]
  const ED = [0, 0.19, 0.31, 0.3932, 0.518, 0.62, 0.71, 0.83, 1.01]
  const SIDE = ['left', 'left', 'right', 'right', 'right', 'left', 'right', 'left']
  const ss = (a, b, t) => { const x = Math.min(1, Math.max(0, (t - a) / (b - a))); return x * x * (3 - 2 * x) }
  const lin = (c) => { c /= 255; return c <= 0.04045 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4) }
  const srgb = (v) => { v = Math.max(0, Math.min(1, v)); return Math.round(255 * (v <= 0.0031308 ? v * 12.92 : 1.055 * Math.pow(v, 1 / 2.4) - 0.055)) }
  const hex = (h) => [1, 3, 5].map((i) => parseInt(h.slice(i, i + 2), 16))
  const mix = (a, b, t) => { const A = hex(a), B = hex(b); return `rgb(${A.map((v, i) => srgb(lin(v) + (lin(B[i]) - lin(v)) * t)).join(',')})` }
  const mixRGB = (a, b, t) => a.map((v, i) => v + (b[i] - v) * t)

  // ── lighting plan: keys [progress, values]; eased between keys
  function plan(p) {
    const keys = K.plan
    if (p <= keys[0][0]) return { ...keys[0][1] }
    for (let i = 1; i < keys.length; i++) {
      if (p <= keys[i][0]) {
        const t = ss(keys[i - 1][0], keys[i][0], p), a = keys[i - 1][1], b = keys[i][1], o = {}
        for (const k in a) o[k] = typeof a[k] === 'number' ? a[k] + (b[k] - a[k]) * t : (t < 0.5 ? a[k] : b[k])
        return o
      }
    }
    return { ...keys[keys.length - 1][1] }
  }

  // ── geometry from the real camera track
  function geo(p, W, H) {
    const c = CH.find((x) => p >= x[1] && p < x[2]) || CH[7]
    const t = Math.min(1, Math.max(0, (p - c[1]) / (c[2] - c[1])))
    const pose = C.cameraAt(c[0], t)
    const mode = C.layoutMode(W, H)
    const L = C.filmLayout(W, H, pose, C.compositionAt(p, mode))
    const S = (q) => { const [fx, fy] = C.projectToFilm(pose, q); return [fx * L.w - L.ox, fy * L.h - L.oy] }
    let hx = pose.target[0] - pose.pos[0], hz = pose.target[2] - pose.pos[2]; const hl = Math.hypot(hx, hz); hx /= hl; hz /= hl
    const pitch = Math.atan2(pose.target[1] - pose.pos[1], Math.hypot(pose.target[0] - pose.pos[0], pose.target[2] - pose.pos[2])) * 180 / Math.PI
    const corners = []
    for (const x of [-0.76, 0.76]) for (const y of [0, 1.8]) for (const z of [-0.69, 0.69]) corners.push(S([x, y, z]))
    const xs = corners.map((q) => q[0]), ys = corners.map((q) => q[1])
    const circle = (cx, cz, R) => { const pts = []; for (let i = 0; i < 40; i++) { const a = (i / 40) * Math.PI * 2; pts.push(S([cx + Math.cos(a) * R, 0, cz + Math.sin(a) * R])) } const X = pts.map((q) => q[0]), Y = pts.map((q) => q[1]); return [(Math.min(...X) + Math.max(...X)) / 2, (Math.min(...Y) + Math.max(...Y)) / 2, (Math.max(...X) - Math.min(...X)) / 2, (Math.max(...Y) - Math.min(...Y)) / 2] }
    // how open the bag is (front panel down / inner box out), for the live shadow footprint
    const open = c[0] === 'interior' ? 1 : c[0] === 'display' ? 1 - ss(0, 0.45, t) : c[0] === 'open' ? ss(0.32, 0.8, t) : 0
    const height = c[0] === 'overview' ? 1.8 - 0.9 * ss(0, 0.55, t) : 1.8
    // photoreal frames covering the live model (sequence chapters, faded at their edges like the site)
    const cover = c[0] === 'intro' ? 1 : (c[0] === 'reveal' || c[0] === 'open') ? ss(0, 0.07, t) * (1 - ss(0.93, 1, t)) : 0
    const ch = ED.findIndex((a, i) => p >= a && p < ED[i + 1]) + 1
    return { c, t, pose, mode, L, S, hx, hz, pitch, bbox: [Math.min(...xs), Math.min(...ys), Math.max(...xs), Math.max(...ys)], cove: S([hx * K.coveDist, 0, hz * K.coveDist])[1], contact: S([0, 0, 0]), pool: circle(0.25, -0.3, 1.55), poolWide: circle(0.2, -0.2, 3.0), open, height, cover, ch, side: mode === 'landscape' ? SIDE[ch - 1] : 'below' }
  }

  // contact shadow for the live chapters: ambient occlusion under the footprint + the key's cast shadow (the key is
  // upper front-left, LightingSystem -2.6, 5.2, 3.6), both blurred like the Cycles shadow in the frames
  function hull(pts) {
    pts = pts.slice().sort((a, b) => a[0] - b[0] || a[1] - b[1])
    const cr = (o, a, b) => (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])
    const lo = [], up = []
    for (const p of pts) { while (lo.length >= 2 && cr(lo[lo.length - 2], lo[lo.length - 1], p) <= 0) lo.pop(); lo.push(p) }
    for (const p of pts.slice().reverse()) { while (up.length >= 2 && cr(up[up.length - 2], up[up.length - 1], p) <= 0) up.pop(); up.push(p) }
    return lo.slice(0, -1).concat(up.slice(0, -1))
  }
  function shadowPolys(g) {
    const zf = 0.69 + 0.75 * g.open, base = [[-0.8, -0.72], [0.8, -0.72], [0.8, zf], [-0.8, zf]]
    const L = K.keyDir || [0.5, -0.69] // floor offset per metre of height
    const foot = base.map(([x, z]) => g.S([x, 0, z]))
    const cast = hull(base.flatMap(([x, z]) => [g.S([x, 0, z]), g.S([x + L[0] * g.height, 0, z + L[1] * g.height])]))
    const aoPts = [[-0.95, -0.85], [0.95, -0.85], [0.95, zf + 0.15], [-0.95, zf + 0.15]].map(([x, z]) => g.S([x, 0, z]))
    return { foot, cast, ao: aoPts, size: Math.abs(g.bbox[2] - g.bbox[0]) }
  }

  // ── layers (inside .stage, before both product canvases, so they always sit behind the product)
  const st = document.querySelector('.stage')
  const env = document.createElement('div'); env.className = 'e-env'
  env.innerHTML = '<i class="e-base"></i>' + (K.window ? '<i class="e-window"></i>' : '') + '<i class="e-wash"></i><i class="e-pool"></i><svg class="e-shadow"></svg><i class="e-side"></i><i class="e-vig"></i>'
  st.insertBefore(env, st.firstChild)
  const svg = env.querySelector('.e-shadow')
  const root = document.documentElement

  let introDoneAt = 0
  const firstLight = () => {
    if (!K.firstLight) return 1
    if (!introDoneAt && document.querySelector('.intro--done')) introDoneAt = performance.now()
    if (!introDoneAt) return 0
    if (matchMedia('(prefers-reduced-motion: reduce)').matches) return 1
    return ss(K.firstLight.delay, K.firstLight.delay + K.firstLight.dur, performance.now() - introDoneAt)
  }
  function update() {
    const W = innerWidth, H = innerHeight
    const spacer = document.querySelector('.narrative-spacer')
    const p = Math.min(1, Math.max(0, scrollY / Math.max(1, spacer.offsetHeight - H)))
    const g = geo(p, W, H), P = plan(p), s = root.style
    const fl1 = firstLight(); if (K.firstLight) { P.wall *= fl1; P.floor *= 0.12 + 0.88 * fl1; P.wash *= 0.4 + 0.6 * fl1 }
    // palette: the same studio at two light levels (and two colour temperatures for the light level)
    const lightPal = {}
    for (const k of ['w0', 'w1', 'cove', 'f0', 'f1']) lightPal[k] = K.lightWarm ? mix(K.light[k], K.lightWarm[k], P.warm || 0) : null
    const pick = (k, lift) => {
      const lp = K.lightWarm ? lightPal[k] : null
      if (!lp) return mix(K.dark[k], K.light[k], lift)
      // mix dark -> (light, warmed)
      const A = hex(K.dark[k]); const B = lp.match(/\d+/g).map(Number)
      return `rgb(${A.map((v, i) => srgb(lin(v) + (lin(B[i]) - lin(v)) * lift)).join(',')})`
    }
    s.setProperty('--e-w0', pick('w0', P.wall)); s.setProperty('--e-w1', pick('w1', P.wall)); s.setProperty('--e-cove', pick('cove', P.wall))
    s.setProperty('--e-f0', pick('f0', P.floor)); s.setProperty('--e-f1', pick('f1', P.floor))
    s.setProperty('--e-cove-y', Math.round(g.cove) + 'px')
    const pool = g.ch === 8 && K.widePoolInSystem ? g.poolWide : g.pool
    const lift = (P.wall + P.floor) / 2
    const poolRGB = mixRGB([232, 228, 221], [255, 253, 249], lift).map(Math.round).join(',')
    s.setProperty('--e-pool', `rgba(${poolRGB},${P.pool.toFixed(3)})`)
    s.setProperty('--e-pool-x', Math.round(pool[0]) + 'px'); s.setProperty('--e-pool-y', Math.round(pool[1]) + 'px')
    s.setProperty('--e-pool-rx', Math.round(pool[2] * P.size) + 'px'); s.setProperty('--e-pool-ry', Math.round(pool[3] * P.size) + 'px')
    const bx = (g.bbox[0] + g.bbox[2]) / 2
    s.setProperty('--e-wash', `rgba(${poolRGB},${P.wash.toFixed(3)})`)
    s.setProperty('--e-wash-x', Math.round(bx) + 'px'); s.setProperty('--e-wash-y', Math.round(Math.min(g.cove, g.bbox[3]) - H * 0.16) + 'px')
    s.setProperty('--e-wash-rx', Math.round(W * (g.ch === 8 ? 0.5 : 0.38)) + 'px'); s.setProperty('--e-wash-ry', Math.round(H * 0.52) + 'px')
    const vig = mixRGB([7, 8, 9], K.lightVig || [110, 104, 96], lift).map(Math.round).join(',')
    s.setProperty('--e-vig', `rgba(${vig},${(P.vig * (1 - 0.45 * lift)).toFixed(3)})`)
    const sideRGB = mixRGB([9, 10, 11], K.lightSide || [214, 210, 202], lift).map(Math.round).join(',')
    s.setProperty('--e-side', `rgba(${sideRGB},${(K.sideA ?? 0.5).toFixed(2)})`)
    const scrim = mixRGB([12, 13, 14], K.lightScrim || [226, 222, 214], lift).map(Math.round).join(',')
    s.setProperty('--e-scrim', scrim)
    root.dataset.side = g.side
    root.dataset.tone = (K.tone ? K.tone(p, P) : (lift > 0.5 ? 'light' : 'dark'))
    root.dataset.ech = g.ch
    // mocked live contact shadow
    const sh = shadowPolys(g), live = (1 - g.cover)
    const blurC = Math.max(6, sh.size * 0.07), blurA = Math.max(3, sh.size * 0.02)
    const shA = (K.shadow ?? 1) * live * (0.55 + 0.45 * lift)
    const P2 = (pts) => pts.map((q) => q.map(Math.round).join(',')).join(' ')
    svg.setAttribute('viewBox', `0 0 ${W} ${H}`)
    svg.innerHTML = shA < 0.01 ? '' : `<defs><filter id="eb1" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="${blurC.toFixed(1)}"/></filter><filter id="eb2" x="-50%" y="-50%" width="200%" height="200%"><feGaussianBlur stdDeviation="${blurA.toFixed(1)}"/></filter>
      <radialGradient id="eg" gradientUnits="userSpaceOnUse" cx="${Math.round(g.contact[0])}" cy="${Math.round(g.contact[1])}" r="${Math.round(sh.size * 1.1)}"><stop offset="0" stop-color="#000" stop-opacity="1"/><stop offset="1" stop-color="#000" stop-opacity="0.25"/></radialGradient></defs>
      <polygon points="${P2(sh.cast)}" fill="url(#eg)" opacity="${(0.34 * shA).toFixed(3)}" filter="url(#eb1)"/>
      <polygon points="${P2(sh.ao)}" fill="#000" opacity="${(0.22 * shA).toFixed(3)}" filter="url(#eb1)"/>
      <polygon points="${P2(sh.foot)}" fill="#000" opacity="${(0.5 * shA).toFixed(3)}" filter="url(#eb2)"/>`
    window.__ENVSTATE = { p, ch: g.ch, plan: P, cover: g.cover, tone: root.dataset.tone }
  }
  const loop = () => { try { update() } catch (e) { window.__ENVERR = String(e && e.stack || e) } requestAnimationFrame(loop) }
  requestAnimationFrame(loop)

  // ── frame-edge extension: continue the photoreal shadow past the film edge instead of ending in a line
  if (K.extend) {
    const orig = CanvasRenderingContext2D.prototype.drawImage
    const probe = document.createElement('canvas'); const pc = probe.getContext('2d', { willReadFrequently: true })
    const edgeAlpha = new WeakMap()
    const maxAlpha = (img, sx, sy, sw, sh) => { probe.width = sw; probe.height = sh; pc.clearRect(0, 0, sw, sh); orig.call(pc, img, sx, sy, sw, sh, 0, 0, sw, sh); const d = pc.getImageData(0, 0, sw, sh).data; let m = 0; for (let i = 3; i < d.length; i += 4) m = Math.max(m, d[i]); return m / 255 }
    CanvasRenderingContext2D.prototype.drawImage = function (img, ...a) {
      orig.call(this, img, ...a)
      if (a.length !== 4 || !this.canvas.classList || !this.canvas.classList.contains('sequence-canvas')) return
      const [x, y, w, h] = a, CW = this.canvas.width, CHh = this.canvas.height
      const iw = img.width, ih = img.height
      let e = edgeAlpha.get(img)
      if (!e) { e = { left: maxAlpha(img, 0, 0, 2, ih), right: maxAlpha(img, iw - 2, 0, 2, ih), bottom: maxAlpha(img, 0, ih - 2, iw, 2) }; edgeAlpha.set(img, e) }
      const ctx = this
      const fade = (x0, y0, x1, y1, rx, ry, rw, rh) => { const gr = ctx.createLinearGradient(x0, y0, x1, y1); gr.addColorStop(0, 'rgba(0,0,0,0)'); gr.addColorStop(1, 'rgba(0,0,0,1)'); ctx.save(); ctx.globalCompositeOperation = 'destination-out'; ctx.fillStyle = gr; ctx.fillRect(rx, ry, rw, rh); ctx.restore() }
      ctx.save(); ctx.imageSmoothingEnabled = true; ctx.imageSmoothingQuality = 'high'
      const fl = Math.min(CW, CHh) * 0.35
      // the edge strip is averaged first (24 source pixels deep, then shrunk across), so the stretched shadow is smooth
      const strip = (sx, sy, sw, sh, ow, oh) => { const t = document.createElement('canvas'); t.width = ow; t.height = oh; const c2 = t.getContext('2d'); c2.imageSmoothingQuality = 'high'; orig.call(c2, img, sx, sy, sw, sh, 0, 0, ow, oh); return t }
      if (y + h < CHh && e.bottom < 0.995) { const t = strip(0, ih - 24, iw, 24, Math.max(8, Math.round(iw / 40)), 1); orig.call(ctx, t, 0, 0, t.width, 1, x, y + h - 1, w, CHh - (y + h) + 1); fade(0, y + h, 0, y + h + fl, x, y + h, w, CHh - (y + h)) }
      if (x + w < CW && e.right < 0.6) { const t = strip(iw - 24, 0, 24, ih, 1, Math.max(8, Math.round(ih / 40))); orig.call(ctx, t, 0, 0, 1, t.height, x + w - 1, y, CW - (x + w) + 1, h); fade(x + w, 0, x + w + fl, 0, x + w, y, CW - (x + w), h) }
      if (x > 0 && e.left < 0.6) { const t = strip(0, 0, 24, ih, 1, Math.max(8, Math.round(ih / 40))); orig.call(ctx, t, 0, 0, 1, t.height, 0, y, x + 1, h); fade(x, 0, x - fl, 0, 0, y, x, h) }
      ctx.restore()
    }
  }
})()
