# Body-text contrast per chapter: the median background behind each chapter's text column (mostly background pixels)
# against the body colour that concept uses there (dark tone #a7b0b8, light tone #434b52). Desktop: the text column;
# phone: the text block under the product.
import os
import numpy as np
from PIL import Image
def lin(c): c = np.asarray(c, float) / 255; return np.where(c <= 0.04045, c / 12.92, ((c + 0.055) / 1.055) ** 2.4)
def L(rgb): r, g, b = lin(rgb); return 0.2126 * r + 0.7152 * g + 0.0722 * b
def cr(a, b): a, b = max(a, b), min(a, b); return (a + 0.05) / (b + 0.05)
SIDE = {'0150': 'left', '0250': 'left', '0380': 'right', '0440': 'right', '0550': 'left', '0660': 'left', '0780': 'right', '0950': 'left'}
CH = {'0150': 1, '0250': 2, '0380': 3, '0440': 4, '0550': 5, '0660': 6, '0780': 7, '0950': 8}
DARK, LIGHT = (167, 176, 184), (53, 60, 66)
def tone(c, ch): return 'light' if c == 'atelier' or (c == 'acts' and 2 <= ch <= 6) else 'dark'
print(f"{'':10}" + ''.join(f'  ch{CH[k]}   ' for k in SIDE))
for dev in ['d', 'm']:
    for c in ['base', 'noir', 'atelier', 'acts']:
        row = []
        for k, side in SIDE.items():
            f = f'out/{c}_{dev}/{k}.png'
            if not os.path.exists(f): row.append('   -   '); continue
            im = np.asarray(Image.open(f).convert('RGB')).astype(float); H, W, _ = im.shape
            if dev == 'd': reg = im[int(H * .3):int(H * .72), *( [slice(int(W * .06), int(W * .38))] if side == 'left' else [slice(int(W * .62), int(W * .94))] )]
            else: reg = im[int(H * .58):int(H * .9), int(W * .06):int(W * .94)]
            reg = reg.reshape(-1, 3); lum = reg @ [0.2126, 0.7152, 0.0722]; med = reg[np.argsort(lum)[len(lum) // 2]]
            body = LIGHT if tone(c, CH[k]) == 'light' else DARK
            row.append(f' {cr(L(body), L(med)):4.1f}:1 ')
        print(f'{c + " " + ("desk" if dev == "d" else "phone"):10}' + ''.join(row))
